# Canonical list of LOCAL secret keys the lead + ops scripts expect in
# the shell environment. Sourced by scripts/load_secrets.sh / .ps1.
#
# One KEY per line. Lines starting with # and blank lines are ignored.
#
# Vault item path convention (both 1Password and Bitwarden):
#   folder / collection: "tinyassets"
#   item name:           the key name (e.g. CLOUDFLARE_API_TOKEN)
#   field:               "password" or "credential" (vendor-specific)
#
# This list intentionally excludes CI-side secrets from loader input. Standing
# CI authorities that need coordinated rotation are documented as commented
# `ROTATION-ONLY` entries so they are visible here without being exported into
# every operator shell or required by load_secrets.sh / .ps1.

# --- Cloudflare ------------------------------------------------------
CLOUDFLARE_API_TOKEN
CLOUDFLARE_ZONE_ID
# CLOUDFLARE_EMAIL and CLOUDFLARE_GLOBAL_KEY are intentionally NOT in
# this list — the scoped API token above replaces them for all ops.
# Global Key usage was retired per HD-4 rotation concern (2026-04-21).

# --- DigitalOcean ----------------------------------------------------
DIGITALOCEAN_TOKEN

# --- Hetzner (legacy fallback host, pre-DO migration) ----------------
# Optional — only needed if operator is debugging the old Hetzner box.
# Loader tolerates absence; `migrate_secrets_to_vault.py` ignores them
# if they're absent from the plaintext file.
HETZNER_EMAIL
HETZNER_PASSWORD

# --- OpenAI / provider keys ------------------------------------------
OPENAI_API_KEY

# --- TinyAssets deployment identity evidence -------------------------
# Dedicated HMAC key for self-only request identity fingerprints.
# Keep separate from OAuth, provider, maintainer, and roster credentials.
TINYASSETS_IDENTITY_FINGERPRINT_KEY

# --- TinyAssets scoped wiki canary authority --------------------------
# ROTATION-ONLY TINYASSETS_WIKI_CANARY_TOKEN
# Replace the production server env value and the GitHub Actions secret
# together. Minimum 32 UTF-8 bytes. This is not OAuth and authorizes only
# drafts/notes/uptime-probe.md. Never load it into general operator shells.

# --- GitHub local push credential ------------------------------------
# Local PAT for operator/sandbox push flows. GitHub Actions' built-in
# GITHUB_TOKEN remains CI-side and is intentionally not listed here.
#
# Required scopes for sandbox bootstrap:
#   contents:write + pull-requests:write
# Add workflows:write only if the sandbox needs to push
# .github/workflows/* edits — otherwise omit to keep blast radius small.
GH_TOKEN

# --- Stripe billing (production) --------------------------------------
# Both live on the daemon in /etc/tinyassets/env (root:tinyassets 0640).
# Neither is baked into compose.yml: compose `environment` values ship in
# the repo, and these are secrets.
#
# STRIPE_SECRET_KEY        sk_live_… — authorizes real charges.
# STRIPE_WEBHOOK_SECRET    whsec_…   — per-ENDPOINT and per-MODE. Swapping
#   test→live means a NEW endpoint and therefore a NEW secret; the webhook
#   refuses any event whose `livemode` disagrees with the key, so a stale
#   test secret fails loudly rather than entitling across modes.
STRIPE_SECRET_KEY
STRIPE_WEBHOOK_SECRET

# --- TinyAssets billing entitlement authority -------------------------
# Signs the HMAC claim in each subscription's Stripe metadata, which is what
# lets a webhook prove a subscription is one WE created for a given universe.
#
# Deliberately NOT the Stripe webhook secret. That one is Stripe's to rotate
# (and must be rotated on any leak); using it here meant rolling it would
# permanently invalidate the claim on every subscription already sold, so
# those subscriptions could never move a tier again. Claims are versioned:
# v1 = webhook secret (legacy, still verified), v2 = this key.
#
# Generate: python -c "import secrets; print(secrets.token_urlsafe(48))"
# ROTATION: rotating this INVALIDATES every v2 subscription. Only rotate with
# a re-signing migration, or by adding a v3 that verifies v2 alongside it.
TINYASSETS_BILLING_ENTITLEMENT_KEY
